START 
    SET marks = [78, 65, 90,54, 88]
    SET max = marks[0]

    FOR  i FROM 1 TO 4 DO 
        IF marks[i] > max THEN 
            SET max = marks[i]
        END IF
    END FOR

    PRINT "the largest marks is :"+ max

END
    
explanation
marks starts as marks[0] = 78 then

i = 1 ---> marks[1] 65 < 78 no change marks stays 78
i = 2 ---> marks[2] 90 > 78 max becomes 90
i = 3 ---> marks[3] 54 < 90 no change max stays 90
i = 4 ---> marks[4] 88 < 90 no cahnge max stays 90

output the largest marks is 90
